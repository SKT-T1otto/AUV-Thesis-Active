class Fake:
    def __init__(self,config,scenario,*,collection_only=False,**kwargs):
        self.step=0;self.terminal=False;self.collection=collection_only;self.safety_transitions=[]
    def advance(self,*args,**kwargs):
        cost=np.array([0,0,0,self.step==7],np.float32)
        row=dict(safety_features=np.full((4,192),self.step/10,np.float32),
            normalized_executed_acceleration=np.full((4,3),-.1,np.float32),collision_cost=cost,
            safety_rows=[dict(mode='NOMINAL_COLLECTION' if self.collection else 'NOMINAL' if self.step>2 else 'BRAKE',model_checked=not self.collection) for _ in range(4)])
        self.safety_transitions.append(row);self.step+=1;self.terminal=self.step==8;return row
    def summary(self):return dict(termination_reason='obstacle_collision',success=False,synthetic_fixture=True)
    def close(self):pass
